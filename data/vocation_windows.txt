-- data/lib/multivocation_config.lua
MULTI_VOC_STORAGE = 90001
MULTI_VOCATIONS = {
  [2] = { -- Sorcerer
    fragment_id = 10010, -- ID do fragmento do sorc
    name = "Sorcerer",
    description = "Mestre dos poderes mágicos ofensivos."
  },
  [3] = { -- Druid
    fragment_id = 10011,
    name = "Druid",
    description = "Especialista em magias de cura e natureza."
  },
  [4] = { -- Paladin
    fragment_id = 10012,
    name = "Paladin",
    description = "Combate a distância e defesa intermediária."
  },
  [5] = { -- Knight
    fragment_id = 10013,
    name = "Knight",
    description = "Maior resistência e combate corpo-a-corpo."
  }
}
REQUIRED_FRAGMENTS = 100
VOCATION_TILE_ACTIONID = 45001 -- Tile especial



-- data/actions/scripts/vocation_fragment.lua
dofile("data/lib/multivocation_config.lua")

function onUse(cid, item, fromPosition, itemEx, toPosition)
  for vocationId, config in pairs(MULTI_VOCATIONS) do
    if item.itemid == config.fragment_id then
      if getPlayerItemCount(cid, item.itemid) < REQUIRED_FRAGMENTS then
        doPlayerSendCancel(cid, "Você precisa de 100 fragments para desbloquear esta vocation.")
        return true
      end
      local storage = getPlayerStorageValue(cid, MULTI_VOC_STORAGE)
      local vocations = storage == -1 and "" or storage
      if vocations:find(","..vocationId..",") then
        doPlayerSendCancel(cid, "Você já desbloqueou esta vocation.")
        return true
      end
      doPlayerRemoveItem(cid, item.itemid, REQUIRED_FRAGMENTS)
      vocations = vocations..","..vocationId..","
      setPlayerStorageValue(cid, MULTI_VOC_STORAGE, vocations)
      doPlayerSendTextMessage(cid, MESSAGE_INFO_DESCR, "Parabéns! Você desbloqueou "..config.name..".")
      return true
    end
  end
  return false
end

-- data/movements/scripts/vocation_tile.lua
dofile("data/lib/multivocation_config.lua")

function onStepIn(cid, item, position, fromPosition)
  local storage = getPlayerStorageValue(cid, MULTI_VOC_STORAGE)
  if storage == -1 or storage == "" then
    doPlayerSendCancel(cid, "Você não possui vocations alternativas desbloqueadas.")
    return true
  end

  local vocations = {}
  for vocationId in storage:gmatch("(%d+)") do
    table.insert(vocations, tonumber(vocationId))
  end

  local modal = ModalWindow(33001, "Escolha sua vocation", "Selecione para ativar:")
  for _, vocationId in ipairs(vocations) do
    local config = MULTI_VOCATIONS[vocationId]
    local info = config and config.description or "Sem descrição."
    modal:addChoice(vocationId, config.name .. " - " .. info)
  end
  modal:addButton(1, "Selecionar")
  modal:addButton(2, "Cancelar")
  modal:setDefaultEnterButton(1)
  modal:setDefaultEscapeButton(2)
  modal:sendToPlayer(cid)
  return true
end

<movevent type="StepIn" actionid="45001" event="script" value="vocation_tile.lua"/>



-- data/creaturescripts/scripts/vocation_modal.lua
function onModalWindow(cid, modalWindowId, buttonId, choiceId)
  if modalWindowId ~= 33001 then return false end
  if buttonId == 1 then -- Selecionar
    if choiceId and tonumber(choiceId) then
      doPlayerSetVocation(cid, tonumber(choiceId))
      doPlayerSendTextMessage(cid, MESSAGE_INFO_DESCR, "Vocation alterada com sucesso!")
    else
      doPlayerSendCancel(cid, "Escolha inválida.")
    end
  else
    doPlayerSendCancel(cid, "Ação cancelada.")
  end
  return true
end
